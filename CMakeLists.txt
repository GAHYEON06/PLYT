cmake_minimum_required(VERSION 3.20)
project(PLYT VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(plyt_sqlite STATIC server/vendor/sqlite3.c)
target_compile_definitions(plyt_sqlite PRIVATE SQLITE_THREADSAFE=1)
add_executable(plyt-server server/main.cpp)
target_link_libraries(plyt-server PRIVATE plyt_sqlite Threads::Threads)
if(WIN32)
  target_compile_definitions(plyt-server PRIVATE _WIN32_WINNT=0x0A00 CPPHTTPLIB_THREAD_POOL_COUNT=8)
  target_link_libraries(plyt-server PRIVATE ws2_32 bcrypt crypt32)
  if(MSVC)
    target_compile_options(plyt-server PRIVATE /utf-8 /EHsc)
  endif()
else()
  find_package(OpenSSL REQUIRED)
  target_compile_definitions(plyt-server PRIVATE CPPHTTPLIB_THREAD_POOL_COUNT=8)
  target_link_libraries(plyt-server PRIVATE OpenSSL::Crypto ${CMAKE_DL_LIBS})
endif()
